    56→function fallbackCopy(text: string, toastFn: (opts: { title: string; variant?: "destructive" }) => void) {
    57→  try {
    58→    const ta = document.createElement("textarea");
    59→    ta.value = text;
    60→    ta.style.position = "fixed";
    61→    ta.style.left = "-9999px";
    62→    document.body.appendChild(ta);
    63→    ta.select();
    64→    document.execCommand("copy");
    65→    document.body.removeChild(ta);
    66→    toastFn({ title: "تم نسخ الرابط" });
    67→  } catch {
    68→    toastFn({ title: "تعذّر نسخ الرابط", variant: "destructive" });
    69→  }
    70→}

   346→    if (navigator.clipboard && navigator.clipboard.writeText) {
   347→      navigator.clipboard.writeText(url).then(() => {
   348→        toast({ title: "تم نسخ الرابط" });

   350→        fallbackCopy(url, toast);
