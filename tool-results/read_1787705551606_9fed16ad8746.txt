   105→// ─── Helpers ───
   106→function extractFieldErrors(body: unknown): Record<string, string> {
   107→  const record: Record<string, string> = {};
   108→  const data = body as Record<string, unknown> | null;
   109→  if (!data || !Array.isArray(data.detail)) return record;
   110→  const details = data.detail as ValidationError[];
   111→  for (const v of details) {
   112→    const fieldName = v.loc[v.loc.length - 1];
   113→    if (typeof fieldName === "string") {
   114→      record[fieldName] = v.msg;
   115→    }
   116→  }
   117→  return record;
   118→}

   126→    image_url: p.image_url ?? "",

   128→    display_order: p.display_order,
   129→  };
   130→}
