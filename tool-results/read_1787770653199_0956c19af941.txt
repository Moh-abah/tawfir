   257→/*  قسم الأقرب إليك — زر جيو + شبكة بالمسافة                            */
   258→/* ------------------------------------------------------------------ */
   259→function NearbySection() {
   260→  const [coords, setCoords] = useState<{
   261→    lat: number;
   262→    lng: number;
   263→  } | null>(null);
   264→  const [locating, setLocating] = useState(false);
   265→  const [enabled, setEnabled] = useState(false);
   266→
   267→  const { data, isLoading, error, refetch } = useNearbyProducts(
   268→    coords?.lat ?? null,
   269→    coords?.lng ?? null,
   270→    10,
   271→    enabled
   272→  );
   273→
   274→  const products = data?.items ?? [];
   275→
   276→  const handleLocate = () => {
   277→    if (typeof navigator === "undefined" || !navigator.geolocation) {
   278→      toast({
   279→        title: "غير مدعوم",
   280→        description: "متصفحك لا يدعم تحديد الموقع.",
   281→        variant: "destructive",
   282→      });
   283→      return;
   284→    }
   285→    setLocating(true);
   286→    navigator.geolocation.getCurrentPosition(
   287→      (pos) => {
   288→        setCoords({ lat: pos.coords.latitude, lng: pos.coords.longitude });
   289→        setEnabled(true);
   290→        setLocating(false);
   291→        toast({ title: "تم تحديد موقعك" });
   292→      },
   293→      (err) => {
   294→        setLocating(false);
   295→        // err.PERMISSION_DENIED = 1, err.POSITION_UNAVAILABLE = 2, err.TIMEOUT = 3
   296→        const friendly =
   297→          err.code === err.PERMISSION_DENIED
   298→            ? "لم نتمكن من تحديد موقعك — يرجى السماح بالوصول إلى الموقع من إعدادات المتصفح."
   299→            : err.code === err.POSITION_UNAVAILABLE
   300→              ? "الموقع غير متاح حالياً — تحقّق من اتصالك بالشبكة أو حاول لاحقاً."
   301→              : err.code === err.TIMEOUT
   302→                ? "انتهت مهلة تحديد الموقع — حاول مرة أخرى."
   303→                : "تعذّر تحديد موقعك. حاول مرة أخرى.";
   304→        toast({
   305→          title: "تعذّر تحديد موقعك",
   306→          description: friendly,
   307→          variant: "destructive",
   308→        });
   309→      },
   310→      { enableHighAccuracy: true, timeout: 10_000, maximumAge: 0 }
   311→    );
   312→  };
   313→
   314→  return (
   315→    <section className="space-y-6" aria-label="الأقرب إليك">
   316→      <div className="flex flex-wrap items-end justify-between gap-3">
   317→        <div className="space-y-1">
   318→          <h2 className="text-xl font-extrabold text-foreground sm:text-2xl">
   319→            الأقرب إليك
   320→          </h2>
   321→          <p className="text-sm text-muted-foreground">
   322→            وجبات على بُعد بضعة كيلومترات من موقعك الحالي
   323→          </p>
   324→        </div>
   325→        <Button
   326→          type="button"
   327→          variant={enabled ? "outline" : "default"}
   328→          onClick={handleLocate}
   329→          disabled={locating}
   330→          className="min-h-[44px] shrink-0 gap-2 rounded-full"
   331→        >
   332→          {locating ? (
   333→            <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   334→          ) : (
   335→            <Navigation className="h-4 w-4" aria-hidden="true" />
   336→          )}
   337→          {enabled ? "إعادة تحديد الموقع" : "حدد موقعي"}
   338→        </Button>
   339→      </div>
   340→
   341→      {!enabled ? (
   342→        <div className="rounded-2xl border border-dashed border-border/60 bg-card p-8 text-center">
   343→          <MapPinned
   344→            className="mx-auto h-10 w-10 text-muted-foreground/50"
   345→            aria-hidden="true"
   346→          />
   347→          <p className="mt-3 text-sm text-muted-foreground">
   348→            اضغط «حدد موقعي» لعرض الوجبات الأقرب إليك مرتبة بالمسافة.
   349→          </p>
   350→        </div>
   351→      ) : error ? (
   352→        <ErrorState
   353→          title="تعذّر تحميل الوجبات القريبة"
   354→          message="حدث خطأ أثناء جلب الوجبات من حولك. حاول مرة أخرى."
   355→          onRetry={() => refetch()}
   356→        />
   357→      ) : isLoading ? (
   358→        <div
   359→          className="grid grid-cols-2 gap-3 sm:grid-cols-3 sm:gap-4 lg:grid-cols-4"
   360→          aria-busy="true"
   361→        >
   362→          {Array.from({ length: 4 }).map((_, i) => (
   363→            <ProductCardSkeleton key={i} />
   364→          ))}
   365→        </div>
   366→      ) : products.length === 0 ? (
   367→        <EmptyState
   368→          icon={MapPinned}
   369→          title="لا توجد وجبات قريبة"
   370→          description="لا توجد وجبات متاحة في نطاق 10 كم من موقعك حالياً."
   371→        />
   372→      ) : (
   373→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 sm:gap-4 lg:grid-cols-4">
   374→          {products.map((p) => (
   375→            <ProductCard key={p.id} product={p} />
   376→          ))}
   377→        </div>
   378→      )}
   379→    </section>
   380→  );
   381→}

   396→          <ImageWithSkeleton
   397→            src={resolveImageUrl(facility.image_url)}
