   212→   212→function FacilityStatsChart({
   213→   213→  facilities,
   214→   214→}: {
   215→   215→  facilities: { id: number; name: string }[];
   216→   216→}) {
   217→   217→  const prefersReduced = useReducedMotion();
   218→   218→
   219→   219→  // يعيد استخدام نفس queryKey الموحّد
   220→   220→  const barQueries = useQueries({
   221→   221→    queries: facilities.map((f) => ({
   222→   222→      queryKey: ["owner-products-stats", f.id],
   223→   223→      queryFn: (): Promise<Paginated<Product>> =>
   224→   224→        ownerService.getOwnerProducts(f.id, { page: 1, page_size: 1 }),
   225→   225→      staleTime: 30_000,

   228→   228→
   229→   229→  const isLoading = barQueries.some((q) => q.isLoading);
   230→   230→
   231→   231→  const facilityData = facilities.map((f, idx) => ({
   232→   232→    name: f.name,
   233→   233→    count: barQueries[idx].data?.total ?? 0,
   234→   234→  }));
   235→   235→
   236→   236→  const maxCount = Math.max(...facilityData.map((d) => d.count), 1);
   237→   237→
   238→   238→  return (
   239→   239→    <Card className="rounded-2xl">
   240→   240→      <CardTitle className="p-5 pb-0 text-lg font-bold">
   241→   241→        أداء المنتجات حسب المنشأة
   242→   242→      </CardTitle>
   243→   243→      <CardContent className="p-5">
   244→   244→        {isLoading ? (
   245→   245→          <div className="space-y-4">
   246→   246→            {facilities.map((f) => (
   247→   247→              <div key={f.id} className="space-y-1.5">
   248→   248→                <Skeleton className="h-4 w-32 rounded" />
   249→   249→                <Skeleton className="h-6 w-full rounded-full" />
   250→   250→              </div>
   251→   251→            ))}
   252→   252→          </div>
   253→   253→        ) : (
   254→   254→          <div className="space-y-4">
   255→   255→            {facilityData.map((d) => {
   256→   256→              const percent = maxCount > 0 ? (d.count / maxCount) * 100 : 0;
   257→   257→              return (
   258→   258→                <div key={d.name}>
   259→   259→                  <motion.div
   260→   260→                    className="h-6 rounded-full bg-primary/60"
   261→   261→                    initial={prefersReduced ? { width: `${percent}%` } : { width: 0 }}
   262→   262→                    animate={{ width: `${percent}%` }}
   263→   263→                    transition={{ duration: 0.6, ease: "easeOut" }}
   264→   264→                  />
   265→   265→                  <div className="mt-1 flex items-center justify-between">
   266→   266→                    <span className="max-w-[70%] truncate text-xs text-muted-foreground">
   267→   267→                      {d.name}
   268→   268→                    </span>
   269→   269→                    <span className="text-xs font-semibold">{d.count}</span>
   270→   270→                  </div>
   271→   271→                </div>
   272→   272→              );
   273→   273→            })}
   274→   274→          </div>
   275→   275→        )}
   276→   276→      </CardContent>
   277→   277→    </Card>
   278→   278→  );

   284→   284→
   285→   285→  const recentQueries = useQueries({
   286→   286→    queries: facilityIds.map((id) => ({
   287→   287→      queryKey: ["recent-products", id],
   288→   288→      queryFn: (): Promise<Paginated<Product>> =>
   289→   289→        ownerService.getOwnerProducts(id, { page: 1, page_size: 5 }),
   290→   290→      staleTime: 30_000,
   291→   291→    })),

   294→   294→  const isLoading = recentQueries.some((q) => q.isLoading);

   299→   299→      (a, b) =>

   312→   312→    ? { initial: { opacity: 1 }, animate: { opacity: 1 } }
   313→   313→    : { initial: { opacity: 0, y: 6 }, animate: { opacity: 1, y: 0 } };
   314→   314→
   315→   315→  return (
   316→   316→    <Card className="rounded-2xl">
   317→   317→      <CardContent className="p-5">
   318→   318→        <div className="mb-4 flex items-center gap-2">
   319→   319→          <Package className="h-5 w-5 text-primary" />

   325→   325→              <div key={i} className="flex items-center gap-3">
   326→   326→                <Skeleton className="h-4 w-28 rounded" />
   327→   327→                <Skeleton className="h-5 w-16 rounded-full" />
   328→   328→                <div className="flex-1" />
   329→   329→                <Skeleton className="h-4 w-16 rounded" />
   330→   330→                <Skeleton className="h-2.5 w-2.5 rounded-full" />
   331→   331→              </div>
   332→   332→            ))}
   333→   333→          </div>
   334→   334→        ) : allProducts.length === 0 ? (
   335→   335→          <p className="py-6 text-center text-sm text-muted-foreground">
   336→   336→            لا توجد منتجات بعد
