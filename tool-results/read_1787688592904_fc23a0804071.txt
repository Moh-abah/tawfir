   254→   254→/* ─── Donut / Ring Chart for card status ────────── */
   255→   255→const RING_RADIUS = 60;
   256→   256→const RING_STROKE = 18;
   257→   257→const RING_CIRCUMFERENCE = 2 * Math.PI * RING_RADIUS;
   258→   258→
   259→   259→interface RingSegment {
   260→   260→  label: string;
   261→   261→  count: number;
   262→   262→  color: string;
   263→   263→  bgColor: string;
   264→   264→}
   265→   265→
   266→   266→function CardStatusRingChart({ segments, total, reduced }: { segments: RingSegment[]; total: number; reduced: boolean }) {
   267→   267→  const segmentData = useMemo(() => {
   268→   268→    const result: (RingSegment & { pct: number; dashLength: number; offset: number })[] = [];
   269→   269→    let runningOffset = 0;
   270→   270→    for (const s of segments) {
   271→   271→      if (s.count <= 0) continue;
   272→   272→      const pct = total > 0 ? s.count / total : 0;
   273→   273→      const dashLength = pct * RING_CIRCUMFERENCE;
   274→   274→      result.push({ ...s, pct, dashLength, offset: runningOffset });
   275→   275→      runningOffset += dashLength;
   276→   276→    }
   277→   277→    return result;
   278→   278→  }, [segments, total]);
   279→   279→
   280→   280→  return (
   281→   281→    <div className="flex flex-col items-center gap-6 sm:flex-row sm:items-center">
   282→   282→      {/* SVG Ring */}
   283→   283→      <div className="relative flex h-40 w-40 shrink-0 items-center justify-center">
   284→   284→        <svg
   285→   285→          width={RING_RADIUS * 2 + RING_STROKE * 2}
   286→   286→          height={RING_RADIUS * 2 + RING_STROKE * 2}
   287→   287→          viewBox={`0 0 ${RING_RADIUS * 2 + RING_STROKE * 2} ${RING_RADIUS * 2 + RING_STROKE * 2}`}
   288→   288→          className="-rotate-90"
   289→   289→        >
   290→   290→          {/* Background track */}
   291→   291→          <circle
   292→   292→            cx={RING_RADIUS + RING_STROKE}
   293→   293→            cy={RING_RADIUS + RING_STROKE}
   294→   294→            r={RING_RADIUS}
   295→   295→            fill="none"
   296→   296→            stroke="currentColor"
   297→   297→            strokeWidth={RING_STROKE}
   298→   298→            className="text-muted/30"
   299→   299→          />
   300→   300→          {/* Segments */}
   301→   301→          {segmentData.map((seg) => (
   302→   302→            <motion.circle
   303→   303→              key={seg.label}
   304→   304→              cx={RING_RADIUS + RING_STROKE}
   305→   305→              cy={RING_RADIUS + RING_STROKE}
   306→   306→              r={RING_RADIUS}
   307→   307→              fill="none"
   308→   308→              stroke={seg.color}
   309→   309→              strokeWidth={RING_STROKE}
   310→   310→              strokeLinecap="round"
   311→   311→              strokeDasharray={`${seg.dashLength} ${RING_CIRCUMFERENCE - seg.dashLength}`}
   312→   312→              initial={reduced ? { strokeDashoffset: -seg.offset } : { strokeDashoffset: RING_CIRCUMFERENCE }}
   313→   313→              animate={{ strokeDashoffset: -seg.offset }}
   314→   314→              transition={{ duration: reduced ? 0 : 0.8, ease: "easeOut" as const }}
   315→   315→            />
   316→   316→          ))}
   317→   317→        </svg>
   318→   318→        {/* Center text */}
   319→   319→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   320→   320→          <span className="text-2xl font-bold">{total}</span>
   321→   321→          <span className="text-xs text-muted-foreground">إجمالي البطاقات</span>
   322→   322→        </div>
   323→   323→      </div>
   324→   324→
   325→   325→      {/* Legend */}
   326→   326→      <div className="flex flex-col gap-3">
   327→   327→        {segments.map((seg) => (
   328→   328→          <div key={seg.label} className="flex items-center gap-3 min-h-[44px]">
   329→   329→            <span
   330→   330→              className="h-3 w-3 shrink-0 rounded-full"
   331→   331→              style={{ backgroundColor: seg.color }}
   332→   332→            />
   333→   333→            <span className="text-sm text-muted-foreground">{seg.label}</span>
   334→   334→            <span className="mr-auto text-sm font-semibold">{seg.count}</span>
   335→   335→          </div>
   336→   336→        ))}
   337→   337→      </div>
   338→   338→    </div>
   339→   339→  );
   340→   340→}
