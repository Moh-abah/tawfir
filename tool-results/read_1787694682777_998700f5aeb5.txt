   255→/*  قسم الأقرب إليك — زر جيو + شبكة بالمسافة                            */
   256→/* ------------------------------------------------------------------ */
   257→function NearbySection() {
   258→  const [coords, setCoords] = useState<{
   259→    lat: number;
   260→    lng: number;
   261→  } | null>(null);
   262→  const [locating, setLocating] = useState(false);
   263→  const [enabled, setEnabled] = useState(false);
   264→
   265→  const { data, isLoading, error, refetch } = useNearbyProducts(
   266→    coords?.lat ?? null,
   267→    coords?.lng ?? null,
   268→    10,
   269→    enabled
   270→  );
   271→
   272→  const products = data?.items ?? [];
   273→
   274→  const handleLocate = () => {
   275→    if (typeof navigator === "undefined" || !navigator.geolocation) {
   276→      toast({
   277→        title: "غير مدعوم",
   278→        description: "متصفحك لا يدعم تحديد الموقع.",
   279→        variant: "destructive",
   280→      });
   281→      return;
   282→    }
   283→    setLocating(true);
   284→    navigator.geolocation.getCurrentPosition(
   285→      (pos) => {
   286→        setCoords({ lat: pos.coords.latitude, lng: pos.coords.longitude });
   287→        setEnabled(true);
   288→        setLocating(false);
   289→        toast({ title: "تم تحديد موقعك" });
   290→      },
   291→      (err) => {
   292→        setLocating(false);
   293→        toast({
   294→          title: "تعذّر تحديد موقعك",
   295→          description: err.message,
   296→          variant: "destructive",
   297→        });
   298→      },
   299→      { enableHighAccuracy: true, timeout: 10_000, maximumAge: 0 }
   300→    );
   301→  };
   302→
   303→  return (
   304→    <section className="space-y-6" aria-label="الأقرب إليك">
   305→      <div className="flex flex-wrap items-end justify-between gap-3">
   306→        <div className="space-y-1">
   307→          <h2 className="text-xl font-extrabold text-foreground sm:text-2xl">
   308→            الأقرب إليك
   309→          </h2>
   310→          <p className="text-sm text-muted-foreground">
   311→            وجبات على بُعد بضعة كيلومترات من موقعك الحالي
   312→          </p>
   313→        </div>
   314→        <Button
   315→          type="button"
   316→          variant={enabled ? "outline" : "default"}
   317→          onClick={handleLocate}
   318→          disabled={locating}
   319→          className="min-h-[44px] shrink-0 gap-2 rounded-full"
   320→        >
   321→          {locating ? (
   322→            <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   323→          ) : (
   324→            <Navigation className="h-4 w-4" aria-hidden="true" />
   325→          )}
   326→          {enabled ? "إعادة تحديد الموقع" : "حدد موقعي"}
   327→        </Button>
   328→      </div>
   329→
   330→      {!enabled ? (
   331→        <div className="rounded-2xl border border-dashed border-border/60 bg-card p-8 text-center">
   332→          <MapPinned
   333→            className="mx-auto h-10 w-10 text-muted-foreground/50"
   334→            aria-hidden="true"
   335→          />
   336→          <p className="mt-3 text-sm text-muted-foreground">
   337→            اضغط «حدد موقعي» لعرض الوجبات الأقرب إليك مرتبة بالمسافة.
   338→          </p>
   339→        </div>
   340→      ) : error ? (
   341→        <ErrorState
   342→          title="تعذّر تحميل الوجبات القريبة"
   343→          message="حدث خطأ أثناء جلب الوجبات من حولك. حاول مرة أخرى."
   344→          onRetry={() => refetch()}
   345→        />
   346→      ) : isLoading ? (
   347→        <div
   348→          className="grid grid-cols-2 gap-3 sm:grid-cols-3 sm:gap-4 lg:grid-cols-4"
   349→          aria-busy="true"
   350→        >
   351→          {Array.from({ length: 4 }).map((_, i) => (
   352→            <ProductCardSkeleton key={i} />
   353→          ))}
   354→        </div>
   355→      ) : products.length === 0 ? (
   356→        <EmptyState
   357→          icon={MapPinned}
   358→          title="لا توجد وجبات قريبة"
   359→          description="لا توجد وجبات متاحة في نطاق 10 كم من موقعك حالياً."
   360→        />
   361→      ) : (
   362→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 sm:gap-4 lg:grid-cols-4">
   363→          {products.map((p) => (
   364→            <ProductCard key={p.id} product={p} />
   365→          ))}
   366→        </div>
   367→      )}
   368→    </section>
   369→  );
   370→}

   375→function FacilityCard({ facility }: { facility: Facility }) {
   376→  const PlaceholderIcon = TYPE_ICON[facility.type];
   377→  const maxDiscount = facility.cards.length
   378→    ? Math.max(...facility.cards.map((c) => c.discount_rate))
   379→    : DISCOUNT_RATE;
   380→
   381→  return (
   382→    <article className="flex flex-col overflow-hidden rounded-xl border border-border/50 bg-card shadow-soft transition-all duration-200 hover:-translate-y-1 hover:shadow-soft-lg">
   383→      <div className="relative aspect-video">
   384→        {facility.image_url ? (
   385→          <ImageWithSkeleton
   386→            src={facility.image_url}
   387→            alt={facility.name}
   388→            fill
   389→            className="h-full w-full"
   390→            skeletonClassName="rounded-none"
   391→          />
   392→        ) : (
   393→          <div
   394→            className="flex h-full w-full items-center justify-center bg-muted"
   395→            role="img"
   396→            aria-label={facility.name}
   397→          >
   398→            <PlaceholderIcon
   399→              className="h-12 w-12 text-muted-foreground/40"
   400→              aria-hidden="true"
   401→            />
   402→          </div>
   403→        )}
   404→        {maxDiscount > 0 && (
   405→          <span className="absolute right-3 top-3 rounded-full bg-accent px-3 py-1.5 text-xs font-extrabold text-accent-foreground shadow-soft">
   406→            خصم حتى {maxDiscount}%
   407→          </span>
   408→        )}
   409→      </div>
   410→      <div className="flex flex-1 flex-col gap-3 p-4">
   411→        <div className="flex items-start justify-between gap-2">
   412→          <h3 className="font-bold leading-snug text-foreground">
   413→            {facility.name}
   414→          </h3>
   415→          <span className="shrink-0 rounded-full bg-secondary/15 px-2.5 py-1 text-[11px] font-bold text-secondary">

   426→          asChild
   427→          variant="outline"
   428→          className="mt-auto min-h-[44px] w-full rounded-full"
   429→        >
   430→          <Link href={`/facilities/${facility.id}`}>
   431→            تصفّح المنتجات
   432→            <ArrowLeft className="h-4 w-4" aria-hidden="true" />
   433→          </Link>
   434→        </Button>
   435→      </div>
   436→    </article>
   437→  );
   438→}
