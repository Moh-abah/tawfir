    44→/** يحدد بداية اليوم الحالي بالمللي ثانية. */
    45→function startOfToday(): number {
    46→  const now = new Date();
    47→  return new Date(
    48→    now.getFullYear(),
    49→    now.getMonth(),
    50→    now.getDate()
    51→  ).getTime();
    52→}

    54→/**
    55→ * بطاقات الإحصاءات الموحّدة — بيانات حقيقية من الـ API فقط.
    56→ * تجمع بين StatsOverview + AnalyticsSection السابقتين.
    57→ * يحلّ مشكلة N+1: طلب واحد لكل منشأة بـ queryKey موحّد.
    58→ */
    59→function StatsOverview({ facilityIds }: { facilityIds: number[] }) {
    60→  const prefersReduced = useReducedMotion();
    61→
    62→  // طلب واحد موحّد لكل منشأة لكل المنتجات — يحلّ N+1
    63→  const totalQueries = useQueries({
    64→    queries: facilityIds.map((id) => ({
    65→      queryKey: ["owner-products-stats", id],
    66→      queryFn: (): Promise<Paginated<Product>> =>
    67→        ownerService.getOwnerProducts(id, { page: 1, page_size: 1 }),
    68→      staleTime: 30_000,
    69→    })),
    70→  });
    71→
    72→  // متاح فقط — يشارك نفس النمط
    73→  const availableQueries = useQueries({
    74→    queries: facilityIds.map((id) => ({
    75→      queryKey: ["owner-available-stats", id],
    76→      queryFn: (): Promise<Paginated<Product>> =>
    77→        ownerService.getOwnerProducts(id, {
    78→          page: 1,
    79→          page_size: 1,
    80→          only_available: true,
    81→        }),
    82→      staleTime: 30_000,
    83→    })),
    84→  });
    85→
    86→  // طلبات كل منشأة — لاستخراج «طلبات اليوم» عبر فلترة محلية
    87→  const ordersQueries = useQueries({
    88→    queries: facilityIds.map((id) => ({
    89→      queryKey: ["owner-orders", id],
    90→      queryFn: (): Promise<OrderListOut[]> =>
    91→        ownerService.getOwnerOrders(id),
    92→      staleTime: 30_000,
    93→    })),
    94→  });
    95→
    96→  const isLoading =
    97→    totalQueries.some((q) => q.isLoading) ||
    98→    ordersQueries.some((q) => q.isLoading);
    99→
   100→  const totalProducts = totalQueries.reduce(
   101→    (sum, q) => sum + (q.data?.total ?? 0),
   102→    0
   103→  );
   104→  const availableProducts = availableQueries.reduce(
   105→    (sum, q) => sum + (q.data?.total ?? 0),
   106→    0
   107→  );
   108→
   109→  // طلبات اليوم (محلياً عبر created_at)
   110→  const todayStart = startOfToday();
   111→  const todayOrders = ordersQueries.reduce((sum, q) => {
   112→    const todays = (q.data ?? []).filter(
   113→      (o) => new Date(o.created_at).getTime() >= todayStart
   114→    ).length;
   115→    return sum + todays;
   116→  }, 0);
   117→  const totalOrders = ordersQueries.reduce(
   118→    (sum, q) => sum + (q.data?.length ?? 0),
   119→    0
   120→  );
   121→
   122→  const stats = [
   123→    {
   124→      id: "facilities",
   125→      icon: Store,
   126→      value: facilityIds.length,
   127→      label: "المنشآت",
   128→      iconBg: "bg-primary/10",
   129→      iconColor: "text-primary",
   130→      borderAccent: "border-l-4 border-l-primary",
   131→    },
   132→    {
   133→      id: "products",
   134→      icon: Package,
   135→      value: isLoading ? null : totalProducts,
   136→      label: "المنتجات",
   137→      iconBg: "bg-secondary/10",
   138→      iconColor: "text-secondary",
   139→      borderAccent: "border-l-4 border-l-secondary",
   140→    },
   141→    {
   142→      id: "today-orders",
   143→      icon: ShoppingBag,
   144→      value: isLoading ? null : todayOrders,
   145→      label: "طلبات اليوم",
   146→      iconBg: "bg-accent/15",
   147→      iconColor: "text-accent-foreground",
   148→      borderAccent: "border-l-4 border-l-accent",
   149→    },
   150→    {
   151→      id: "available",
   152→      icon: CheckCircle2,
   153→      value: isLoading ? null : availableProducts,
   154→      label: "متاح",
   155→      iconBg: "bg-secondary/10",
   156→      iconColor: "text-secondary",
   157→      borderAccent: "border-l-4 border-l-secondary",
   158→    },
   159→  ];
   160→
   161→  const containerVariants = prefersReduced
   162→    ? { initial: { opacity: 1 }, animate: { opacity: 1 } }
   163→    : {
   164→        initial: { opacity: 0, y: 12 },
   165→        animate: { opacity: 1, y: 0 },
   166→      };
   167→
   168→  return (
   169→    <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   170→      {stats.map((stat, idx) => {
   171→        const Icon = stat.icon;
   172→        return (
   173→          <motion.div
   174→            key={stat.id}
   175→            variants={containerVariants}
   176→            initial="initial"
   177→            animate="animate"
   178→            transition={{ duration: 0.3, delay: idx * 0.1 }}
   179→          >
   180→            <Card className={cn("rounded-2xl", stat.borderAccent)}>
   181→              <CardContent className="flex items-center gap-4 p-5">
   182→                <div
   183→                  className={cn(
   184→                    "flex h-12 w-12 shrink-0 items-center justify-center rounded-full",
   185→                    stat.iconBg
   186→                  )}
   187→                >
   188→                  <Icon className={cn("h-6 w-6", stat.iconColor)} />
   189→                </div>
   190→                <div>
   191→                  {stat.value === null ? (
   192→                    <Skeleton className="mb-1 h-8 w-12 rounded" />
   193→                  ) : (
   194→                    <p className="text-2xl font-bold leading-none">
   195→                      {stat.value}
   196→                    </p>
   197→                  )}
   198→                  <p className="mt-1 text-sm text-muted-foreground">
   199→                    {stat.label}
   200→                  </p>
   201→                </div>
   202→              </CardContent>
   203→            </Card>
   204→          </motion.div>
   205→        );
   206→      })}
   207→    </div>

   211→// ─── Facility Stats Mini Chart ───
   212→function FacilityStatsChart({
   213→  facilities,
   214→}: {
   215→  facilities: { id: number; name: string }[];
   216→}) {
   217→  const prefersReduced = useReducedMotion();
   218→
   219→  // يعيد استخدام نفس queryKey الموحّد
   220→  const barQueries = useQueries({
   221→    queries: facilities.map((f) => ({
   222→      queryKey: ["owner-products-stats", f.id],
   223→      queryFn: (): Promise<Paginated<Product>> =>
   224→        ownerService.getOwnerProducts(f.id, { page: 1, page_size: 1 }),
   225→      staleTime: 30_000,
   226→    })),
   227→  });
   228→
   229→  const isLoading = barQueries.some((q) => q.isLoading);
   230→
   231→  const facilityData = facilities.map((f, idx) => ({
   232→    name: f.name,
   233→    count: barQueries[idx].data?.total ?? 0,
   234→  }));
   235→
   236→  const maxCount = Math.max(...facilityData.map((d) => d.count), 1);
   237→
   238→  return (
   239→    <Card className="rounded-2xl">
   240→      <CardTitle className="p-5 pb-0 text-lg font-bold">
   241→        أداء المنتجات حسب المنشأة
   242→      </CardTitle>
   243→      <CardContent className="p-5">
   244→        {isLoading ? (
   245→          <div className="space-y-4">
   246→            {facilities.map((f) => (
   247→              <div key={f.id} className="space-y-1.5">
   248→                <Skeleton className="h-4 w-32 rounded" />
   249→                <Skeleton className="h-6 w-full rounded-full" />
   250→              </div>
   251→            ))}
   252→          </div>
   253→        ) : (
   254→          <div className="space-y-4">
   255→            {facilityData.map((d) => {
   256→              const percent = maxCount > 0 ? (d.count / maxCount) * 100 : 0;
   257→              return (
   258→                <div key={d.name}>
   259→                  <motion.div
   260→                    className="h-6 rounded-full bg-primary/60"
   261→                    initial={prefersReduced ? { width: `${percent}%` } : { width: 0 }}
   262→                    animate={{ width: `${percent}%` }}
   263→                    transition={{ duration: 0.6, ease: "easeOut" }}
   264→                  />
   265→                  <div className="mt-1 flex items-center justify-between">
   266→                    <span className="max-w-[70%] truncate text-xs text-muted-foreground">
   267→                      {d.name}
   268→                    </span>
   269→                    <span className="text-xs font-semibold">{d.count}</span>
   270→                  </div>
   271→                </div>
   272→              );
   273→            })}
   274→          </div>
   275→        )}
   276→      </CardContent>
   277→    </Card>
   278→  );
   279→}

   281→// ─── Recent Products Widget (real data from API) ───
   282→function RecentProductsWidget({ facilityIds }: { facilityIds: number[] }) {
   283→  const prefersReduced = useReducedMotion();
   284→
   285→  const recentQueries = useQueries({
   286→    queries: facilityIds.map((id) => ({
   287→      queryKey: ["recent-products", id],
   288→      queryFn: (): Promise<Paginated<Product>> =>
   289→        ownerService.getOwnerProducts(id, { page: 1, page_size: 5 }),
   290→      staleTime: 30_000,
   291→    })),
   292→  });
   293→
   294→  const isLoading = recentQueries.some((q) => q.isLoading);
   295→
   296→  const allProducts = recentQueries
   297→    .flatMap((q) => q.data?.items ?? [])
   298→    .sort(
   299→      (a, b) =>
   300→        new Date(b.created_at).getTime() - new Date(a.created_at).getTime()
   301→    )
   302→    .slice(0, 5);
   303→
   304→  const staggerContainer = prefersReduced
   305→    ? { initial: { opacity: 1 }, animate: { opacity: 1 } }
   306→    : {
   307→        initial: { opacity: 0 },
   308→        animate: { opacity: 1, transition: { staggerChildren: 0.06 } },
   309→      };
   310→
   311→  const staggerItem = prefersReduced
   312→    ? { initial: { opacity: 1 }, animate: { opacity: 1 } }
   313→    : { initial: { opacity: 0, y: 6 }, animate: { opacity: 1, y: 0 } };
   314→
   315→  return (
   316→    <Card className="rounded-2xl">
   317→      <CardContent className="p-5">
   318→        <div className="mb-4 flex items-center gap-2">
   319→          <Package className="h-5 w-5 text-primary" />
   320→          <h2 className="text-lg font-bold">أحدث المنتجات</h2>
   321→        </div>
   322→        {isLoading ? (
   323→          <div className="space-y-3">
   324→            {[1, 2, 3, 4, 5].map((i) => (
   325→              <div key={i} className="flex items-center gap-3">
   326→                <Skeleton className="h-4 w-28 rounded" />
   327→                <Skeleton className="h-5 w-16 rounded-full" />
   328→                <div className="flex-1" />
   329→                <Skeleton className="h-4 w-16 rounded" />
   330→                <Skeleton className="h-2.5 w-2.5 rounded-full" />
   331→              </div>
   332→            ))}
   333→          </div>
   334→        ) : allProducts.length === 0 ? (
   335→          <p className="py-6 text-center text-sm text-muted-foreground">
   336→            لا توجد منتجات بعد
   337→          </p>
   338→        ) : (
   339→          <motion.div
   340→            className="divide-y"
   341→            variants={staggerContainer}
   342→            initial="initial"
   343→            animate="animate"
   344→          >
   345→            {allProducts.map((product) => (
   346→              <motion.div
   347→                key={product.id}
   348→                variants={staggerItem}
   349→                className="flex items-center gap-3 py-3 first:pt-0 last:pb-0"
   350→              >
   351→                <div className="min-w-0 flex-1">
   352→                  <p className="truncate text-sm font-medium">{product.name}</p>
   353→                </div>
   354→                <Badge variant="secondary" className="shrink-0 text-xs">
   355→                  {product.category}
   356→                </Badge>
   357→                <span
   358→                  className={cn(
   359→                    "inline-block h-2.5 w-2.5 shrink-0 rounded-full",
   360→                    product.is_available
   361→                      ? "bg-success"
   362→                      : "bg-muted-foreground/40"
   363→                  )}
   364→                />
   365→                <span className="shrink-0 font-mono text-sm">
   366→                  {formatCurrency(product.price)}
   367→                </span>
   368→              </motion.div>
   369→            ))}
   370→          </motion.div>
   371→        )}
   372→      </CardContent>
   373→    </Card>
   374→  );
   375→}
