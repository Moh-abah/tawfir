   256→/*  قسم الأقرب إليك — زر جيو + شبكة بالمسافة                            */
   257→/* ------------------------------------------------------------------ */
   258→function NearbySection() {
   259→  const [coords, setCoords] = useState<{
   260→    lat: number;
   261→    lng: number;
   262→  } | null>(null);
   263→  const [locating, setLocating] = useState(false);
   264→  const [enabled, setEnabled] = useState(false);
   265→
   266→  const { data, isLoading, error, refetch } = useNearbyProducts(
   267→    coords?.lat ?? null,
   268→    coords?.lng ?? null,
   269→    10,
   270→    enabled
   271→  );
   272→
   273→  const products = data?.items ?? [];
   274→
   275→  const handleLocate = () => {
   276→    if (typeof navigator === "undefined" || !navigator.geolocation) {
   277→      toast({
   278→        title: "غير مدعوم",
   279→        description: "متصفحك لا يدعم تحديد الموقع.",
   280→        variant: "destructive",
   281→      });
   282→      return;
   283→    }
   284→    setLocating(true);
   285→    navigator.geolocation.getCurrentPosition(
   286→      (pos) => {
   287→        setCoords({ lat: pos.coords.latitude, lng: pos.coords.longitude });
   288→        setEnabled(true);
   289→        setLocating(false);
   290→        toast({ title: "تم تحديد موقعك" });
   291→      },
   292→      (err) => {
   293→        setLocating(false);
   294→        // err.PERMISSION_DENIED = 1, err.POSITION_UNAVAILABLE = 2, err.TIMEOUT = 3
   295→        const friendly =
   296→          err.code === err.PERMISSION_DENIED
   297→            ? "لم نتمكن من تحديد موقعك — يرجى السماح بالوصول إلى الموقع من إعدادات المتصفح."
   298→            : err.code === err.POSITION_UNAVAILABLE
   299→              ? "الموقع غير متاح حالياً — تحقّق من اتصالك بالشبكة أو حاول لاحقاً."
   300→              : err.code === err.TIMEOUT
   301→                ? "انتهت مهلة تحديد الموقع — حاول مرة أخرى."
   302→                : "تعذّر تحديد موقعك. حاول مرة أخرى.";
   303→        toast({
   304→          title: "تعذّر تحديد موقعك",
   305→          description: friendly,
   306→          variant: "destructive",
   307→        });
   308→      },
   309→      { enableHighAccuracy: true, timeout: 10_000, maximumAge: 0 }
   310→    );
   311→  };
   312→
   313→  return (
   314→    <section className="space-y-6" aria-label="الأقرب إليك">
   315→      <div className="flex flex-wrap items-end justify-between gap-3">
   316→        <div className="space-y-1">
   317→          <h2 className="text-xl font-extrabold text-foreground sm:text-2xl">
   318→            الأقرب إليك
   319→          </h2>
   320→          <p className="text-sm text-muted-foreground">
   321→            وجبات على بُعد بضعة كيلومترات من موقعك الحالي
   322→          </p>
   323→        </div>
   324→        <Button
   325→          type="button"
   326→          variant={enabled ? "outline" : "default"}
   327→          onClick={handleLocate}
   328→          disabled={locating}
   329→          className="min-h-[44px] shrink-0 gap-2 rounded-full"
   330→        >
   331→          {locating ? (
   332→            <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   333→          ) : (
   334→            <Navigation className="h-4 w-4" aria-hidden="true" />
   335→          )}
   336→          {enabled ? "إعادة تحديد الموقع" : "حدد موقعي"}
   337→        </Button>
   338→      </div>
   339→
   340→      {!enabled ? (
   341→        <div className="rounded-2xl border border-dashed border-border/60 bg-card p-8 text-center">
   342→          <MapPinned
   343→            className="mx-auto h-10 w-10 text-muted-foreground/50"
   344→            aria-hidden="true"
   345→          />
   346→          <p className="mt-3 text-sm text-muted-foreground">
   347→            اضغط «حدد موقعي» لعرض الوجبات الأقرب إليك مرتبة بالمسافة.
   348→          </p>
   349→        </div>
   350→      ) : error ? (
   351→        <ErrorState
   352→          title="تعذّر تحميل الوجبات القريبة"
   353→          message="حدث خطأ أثناء جلب الوجبات من حولك. حاول مرة أخرى."
   354→          onRetry={() => refetch()}
   355→        />
   356→      ) : isLoading ? (
   357→        <div
   358→          className="grid grid-cols-2 gap-3 sm:grid-cols-3 sm:gap-4 lg:grid-cols-4"
   359→          aria-busy="true"
   360→        >
   361→          {Array.from({ length: 4 }).map((_, i) => (
   362→            <ProductCardSkeleton key={i} />
   363→          ))}
   364→        </div>
   365→      ) : products.length === 0 ? (
   366→        <EmptyState
   367→          icon={MapPinned}
   368→          title="لا توجد وجبات قريبة"
   369→          description="لا توجد وجبات متاحة في نطاق 10 كم من موقعك حالياً."
   370→        />
   371→      ) : (
   372→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 sm:gap-4 lg:grid-cols-4">
   373→          {products.map((p) => (
   374→            <ProductCard key={p.id} product={p} />
   375→          ))}
   376→        </div>
   377→      )}
   378→    </section>
   379→  );
   380→}
