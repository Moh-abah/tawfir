   361→/* ─── Donut / Ring Chart for card status ────────── */
   362→const RING_RADIUS = 60;
   363→const RING_STROKE = 18;
   364→const RING_CIRCUMFERENCE = 2 * Math.PI * RING_RADIUS;
   365→
   366→interface RingSegment {
   367→  label: string;
   368→  count: number;
   369→  color: string;
   370→  bgColor: string;
   371→}
   372→
   373→function CardStatusRingChart({ segments, total, reduced }: { segments: RingSegment[]; total: number; reduced: boolean }) {
   374→  const segmentData = useMemo(() => {
   375→    const result: (RingSegment & { pct: number; dashLength: number; offset: number })[] = [];
   376→    let runningOffset = 0;
   377→    for (const s of segments) {
   378→      if (s.count <= 0) continue;
   379→      const pct = total > 0 ? s.count / total : 0;
   380→      const dashLength = pct * RING_CIRCUMFERENCE;
   381→      result.push({ ...s, pct, dashLength, offset: runningOffset });
   382→      runningOffset += dashLength;
   383→    }
   384→    return result;
   385→  }, [segments, total]);
   386→
   387→  return (
   388→    <div className="flex flex-col items-center gap-6 sm:flex-row sm:items-center">
   389→      {/* SVG Ring */}
   390→      <div className="relative flex h-40 w-40 shrink-0 items-center justify-center">
   391→        <svg
   392→          width={RING_RADIUS * 2 + RING_STROKE * 2}
   393→          height={RING_RADIUS * 2 + RING_STROKE * 2}
   394→          viewBox={`0 0 ${RING_RADIUS * 2 + RING_STROKE * 2} ${RING_RADIUS * 2 + RING_STROKE * 2}`}
   395→          className="-rotate-90"
   396→        >
   397→          {/* Background track */}
   398→          <circle
   399→            cx={RING_RADIUS + RING_STROKE}
   400→            cy={RING_RADIUS + RING_STROKE}
   401→            r={RING_RADIUS}
   402→            fill="none"
   403→            stroke="currentColor"
   404→            strokeWidth={RING_STROKE}
   405→            className="text-muted/30"
   406→          />
   407→          {/* Segments */}
   408→          {segmentData.map((seg) => (
   409→            <motion.circle
   410→              key={seg.label}
   411→              cx={RING_RADIUS + RING_STROKE}
   412→              cy={RING_RADIUS + RING_STROKE}
   413→              r={RING_RADIUS}
   414→              fill="none"
   415→              stroke={seg.color}
   416→              strokeWidth={RING_STROKE}
   417→              strokeLinecap="round"
   418→              strokeDasharray={`${seg.dashLength} ${RING_CIRCUMFERENCE - seg.dashLength}`}
   419→              initial={reduced ? { strokeDashoffset: -seg.offset } : { strokeDashoffset: RING_CIRCUMFERENCE }}
   420→              animate={{ strokeDashoffset: -seg.offset }}
   421→              transition={{ duration: reduced ? 0 : 0.8, ease: "easeOut" as const }}
   422→            />
   423→          ))}
   424→        </svg>
   425→        {/* Center text */}
   426→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   427→          <span className="text-2xl font-bold">{total}</span>
   428→          <span className="text-xs text-muted-foreground">إجمالي البطاقات</span>
   429→        </div>
   430→      </div>
   431→
   432→      {/* Legend */}
   433→      <div className="flex flex-col gap-3">
   434→        {segments.map((seg) => (
   435→          <div key={seg.label} className="flex items-center gap-3 min-h-[44px]">
   436→            <span
   437→              className="h-3 w-3 shrink-0 rounded-full"
   438→              style={{ backgroundColor: seg.color }}
   439→            />
   440→            <span className="text-sm text-muted-foreground">{seg.label}</span>
   441→            <span className="mr-auto text-sm font-semibold">{seg.count}</span>
   442→          </div>
   443→        ))}
   444→      </div>
   445→    </div>
   446→  );
   447→}
